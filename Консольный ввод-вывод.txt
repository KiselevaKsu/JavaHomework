Работа с консолью осуществляется с помощью класса System и его статических полей in, out и err.

// Печать данных в консоли
System.out.println('Привет!');
// Ввод символа из консоли
// Требуется обработка исключительной ситуации
try {
    int x = System.in.read();
} catch (IOException e) {
    // обработка ошибки
}
